<!-- Script for copying Youtube Playlist videos -->

(async ()=> {
  const TOOL= 'Export playlist';
  const FORMAT= 'csv';        
  
  if (location.hostname !== 'www.youtube.com' || location.pathname !== '/playlist') {
    alert('Open a playlist first, for example:\nhttps://www.youtube.com/playlist?list=WL');
    return;
  }
  const many= (n, word)=> `${n} ${word}${n=== 1 ? '' : 's'}`;   // 1 video, 2 videos
  const panel= (()=> {
    const box= document.createElement('div');
    box.style.cssText= 'display:block!important;position:fixed;right:16px;bottom:16px;z-index:2147483647;width:340px;max-width:calc(100vw - 32px);' +
      'background:#111318;color:#e6e8ee;font:14px/1.5 system-ui,sans-serif;border-radius:12px;padding:16px;' +
      'box-shadow:0 8px 30px rgba(0,0,0,.4);text-align:left';
    const head= document.createElement('strong');
    head.textContent= TOOL;
    const msg= document.createElement('p');
    msg.style.cssText= 'margin:8px 0 12px;white-space:pre-line';
    const row= document.createElement('div');
    row.style.cssText= 'display:flex;flex-wrap:wrap;gap:8px;justify-content:flex-end';
    box.append(head, msg, row);
    document.body.append(box);
    const button= (label, primary, onClick)=> {
      const b= document.createElement('button');
      b.textContent= label;
      b.style.cssText= 'font:600 14px system-ui,sans-serif;padding:8px 14px;border-radius:8px;cursor:pointer;' +
        (primary ? 'background:#0f766e;color:#fff;border:1px solid #0f766e' : 'background:none;color:#e6e8ee;border:1px solid #3a3f4b');
      b.onclick= onClick;
      row.append(b);
    };
    return {
      text(t) { msg.textContent= t; },
      ask(t, yes) {
        msg.textContent= t;
        row.replaceChildren();
        return new Promise(r=> {
          button('Cancel', false, ()=> { row.replaceChildren(); r(false); });
          button(yes, true, ()=> { row.replaceChildren(); r(true); });
        });
      },
      stoppable(onStop) { row.replaceChildren(); button('Stop', false, onStop); },
      actions(t, list) {
        msg.textContent= t;
        row.replaceChildren();
        list.forEach(([label, primary, run])=> button(label, primary, run));
      },
      finish(t) { this.actions(t, [['Close', true, ()=> box.remove()]]); },
      close() { box.remove(); },
    };
  })();
  const deliver= (summary, body, file)=> {
    const save= ()=> {
      const a= document.createElement('a');
      a.href= URL.createObjectURL(new Blob([/\.csv$/.test(file) ? '\ufeff' + body : body], { type: 'text/plain;charset=utf-8' }));
      a.download= file;
      a.click();
      setTimeout(()=> URL.revokeObjectURL(a.href), 5000);
      panel.text(`${summary}\nSaved as ${file} (check your Downloads).`);
    };
    const copy= ()=> navigator.clipboard.writeText(body)
      .then(()=> panel.text(`${summary}\nCopied. Paste it into a spreadsheet or a note.`))
      .catch(()=> panel.text(`${summary}\nThe browser blocked the clipboard: use Download instead.`));
    panel.actions(summary, [['Close', false, ()=> panel.close()], ['Copy', false, copy], ['Download', true, save]]);
  };
  const sleep= ms=> new Promise(r=> setTimeout(r, ms));
  const rows= ()=> [...document.querySelectorAll('yt-lockup-view-model, ytd-playlist-video-renderer')]
    .filter(el=> el.querySelector('a[href*="/watch?v="][href*="index="]'));

  let last= -1, idle= 0;
  while (idle < 3) {
    window.scrollTo(0, document.documentElement.scrollHeight);
    await sleep(1200);
    const n= rows().length;
    idle= n=== last ? idle + 1 : 0;
    last= n;
    panel.text(`Loading the playlist… ${n} videos`);
  }

  const seen= new Set();
  const videos= [];
  for (const el of rows()) {
    const link= new URL(el.querySelector('a[href*="/watch?v="]').href);
    const id= link.searchParams.get('v');
    const index= link.searchParams.get('index') || '';
    if (seen.has(id + index)) continue;
    seen.add(id + index);
    const text= sel=> (el.querySelector(sel)?.textContent || '').trim();
    videos.push({
      index,
      title: el.querySelector('h3')?.getAttribute('title') || text('#video-title'),
      channel: text('a[href^="/@"], a[href^="/channel/"], ytd-channel-name a'),
      duration: text('.ytBadgeShapeText, ytd-thumbnail-overlay-time-status-renderer #text'),
      url: 'https://www.youtube.com/watch?v=' + id,
    });
  }
  if (!videos.length) {
    panel.finish('No videos found. YouTube may have changed its layout: please report it on responsive-muse.com.');
    return 0;
  }

  const cell= v=> '"' + String(v).replace(/"/g, '""') + '"';
  const body= FORMAT=== 'txt'
    ? videos.map(v=> v.url).join('\n')
    : ['Index,Title,Channel,Duration,URL', ...videos.map(v=>
        [v.index, v.title, v.channel, v.duration, v.url].map(cell).join(','))].join('\n');
  const name= (document.title.replace(/ - YouTube$/, '') || 'playlist').replace(/[\\/:*?"<>|]+/g, '_');
  deliver(`Found ${many(videos.length, 'video')}.`, body, `${name}.${FORMAT}`);
  return videos.length;
})();
